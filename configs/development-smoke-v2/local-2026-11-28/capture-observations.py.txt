def gguf_metadata(path):
    formats = {0:'B',1:'b',2:'H',3:'h',4:'I',5:'i',6:'f',7:'?',10:'Q',11:'q',12:'d'}
    with path.open('rb') as f:
        def unpack(fmt):
            return struct.unpack('<'+fmt, f.read(struct.calcsize('<'+fmt)))[0]
        def string():
            return f.read(unpack('Q')).decode('utf8')
        def value(kind, keep=True):
            if kind == 8:
                size=unpack('Q')
                if keep:
                    return f.read(size).decode('utf8')
                f.seek(size, 1)
                return None
            if kind == 9:
                subtype, size=unpack('I'), unpack('Q')
                if subtype in formats:
                    f.seek(struct.calcsize('<'+formats[subtype])*size, 1)
                else:
                    for _ in range(size): value(subtype, False)
                return {'array_type':subtype,'count':size}
            return unpack(formats[kind])
        assert f.read(4)==b'GGUF'
        version=unpack('I'); tensors=unpack('Q'); count=unpack('Q')
        result={'gguf_version':version,'tensor_count':tensors,'metadata':{}}
        for _ in range(count):
            key=string(); kind=unpack('I'); start=f.tell()
            item=value(kind)
            end=f.tell()
            if key.startswith(('general.', 'tokenizer.')) or key.endswith('.context_length'):
                if kind==9:
                    f.seek(start); item['sha256']=hashlib.sha256(f.read(end-start)).hexdigest(); f.seek(end)
                result['metadata'][key]=item
        return result

async def count_qwen():
    async with lmstudio.AsyncClient('127.0.0.1:1234') as client:
        loaded=await client.llm.list_loaded()
        model=next(m for m in loaded if m.identifier=='qwen2.5-coder-14b-instruct')
        smoke,target=ready['qwen']
        records=[]
        sem=asyncio.Semaphore(4)
        async def one(job):
            async with sem:
                formatted=await model.apply_prompt_template({'messages':job.request['body']['messages']})
                tokens=await model.tokenize(formatted)
                return {'case_uid':job.case.case_id,'strategy':job.strategy,'request_sha256':digest(job.request),'formatted_prompt_sha256':hashlib.sha256(formatted.encode()).hexdigest(),'template_only_tokens':len(tokens)}
        for start in range(0,len(target.jobs),128):
            records.extend(await asyncio.gather(*(one(j) for j in target.jobs[start:start+128])))
            print('template tokenization',len(records),'/',len(target.jobs),flush=True)
        selected={digest(j.request) for j in smoke.jobs}
        save(RAW/'qwen.template-only-counts.json',records)
        summary={'checked_at':datetime.now(UTC).isoformat(),'method':'LM Studio SDK 1.5.0 apply_prompt_template + tokenize, existing loaded instance only','measurement_scope':'template-only diagnostic; NOT InfrastructureInputAccounting; schema contribution to OpenAI HTTP prompt not verified','records_path':'bundle/local-2026-11-28/qwen.template-only-counts.json','records_sha256':sha(RAW/'qwen.template-only-counts.json'),'target_requests_sha256':target.identity['requests_sha256'],'smoke_requests_sha256':smoke.identity['requests_sha256'],'target_count':len(records),'smoke_count':sum(r['request_sha256'] in selected for r in records),'strategies':{}}
        for strategy in ('direct','z3'):
            full=[r for r in records if r['strategy']==strategy]
            sub=[r for r in full if r['request_sha256'] in selected]
            summary['strategies'][strategy]={'target_max':max(r['template_only_tokens'] for r in full),'smoke_max':max(r['template_only_tokens'] for r in sub),'target_max_records':[r for r in full if r['template_only_tokens']==max(x['template_only_tokens'] for x in full)],'smoke_max_records':[r for r in sub if r['template_only_tokens']==max(x['template_only_tokens'] for x in sub)]}
        save(OUT/'qwen.template-only-summary.json',summary)
