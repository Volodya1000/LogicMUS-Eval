def save(path,data):path.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n',encoding='utf8',newline='\n')

def main():
    freeze=json.loads((OUT/'smoke.execution.freeze.json').read_text())
    assert compute_source_bundle_hash()==freeze['source_bundle_sha256']
    for rel,sha in freeze['files'].items():assert compute_file_sha256(ROOT/rel)==sha,rel
    protected=json.loads((OUT/'source-preservation.json').read_text())['protected_files']
    assert all(compute_file_sha256(REPO/rel)==sha for rel,sha in protected.items())
    candidate=REPO/'releases/confirmation-v2-candidate'
    candidate_index=json.loads((candidate/'artifact-index.json').read_text())['files']
    assert all((candidate/rel).stat().st_size==r['bytes'] and compute_file_sha256(candidate/rel)==r['sha256'] for rel,r in candidate_index.items())
    ledger=[json.loads(l) for l in (RAW/'attempt-ledger.jsonl').read_text().splitlines()]
    intents=[r for r in ledger if r['event']=='intent'];diag=[r for r in intents if r['phase']=='diagnostic'];smoke=[r for r in intents if r['phase']=='smoke']
    assert len(intents)<=96 and len(diag)<=12 and len(smoke)<=84
    assert all(r['purpose']=='development' for r in intents)
    assert len({r['request_sha256'] for r in smoke})==len(smoke)
    measured=[];summaries={};executions={};checks={}
    for name in NAMES:
        run=RAW/(name+'.run');snapshot=read_snapshot(run,[run/'report.json'])
        ready=preflight(load_config(OUT/(name+'.smoke.json')))
        assert snapshot.manifest.payload.run_id==freeze['runs'][name]['run_id']
        assert all(len(s.attempts)==1 and s.outcome is not None for s in snapshot.states.values())
        for state in snapshot.states.values():
            for attempt in state.attempts:
                assert attempt.intent.invocation_origin=='http-adapter'
                measured.append((snapshot.manifest.payload.run_id,attempt.intent.request_sha256))
        result=json.loads((OUT/(name+'.smoke-results.json')).read_text());summaries[name]=result
        executions[name]=json.loads((OUT/(name+'.execution.json')).read_text())
        assert executions[name]['noop_resume_http_attempts']==0
        backend=json.loads((OUT/(name+'.smoke-backend.before.json')).read_text())
        evidence_path=OUT/(name+'.evidence.json')
        save(OUT/(name+'.backend-execution-link.json'),{'run_id':snapshot.manifest.payload.run_id,'instance_reference':backend['info']['instanceReference'],'served_model_id':backend['info']['identifier'],'base_config_sha256':backend['base_config_sha256'],'observed_backend_before_sha256':compute_file_sha256(OUT/(name+'.smoke-backend.before.json')),'execution_receipt_sha256':compute_file_sha256(OUT/(name+'.execution.json')),'execution_freeze_sha256':compute_file_sha256(OUT/'smoke.execution.freeze.json'),'smoke_evidence_sha256':compute_file_sha256(evidence_path) if evidence_path.exists() else None,'real_smoke':result['real_smoke'],'evidence_class':'automated observed metadata + production journal verification; no human attestation'})
        command=[sys.executable,'-m','logicmus_eval.cli','check-development-smoke','--root',str(ROOT),'--config',str(OUT/(name+'.smoke.json')),'--evidence',str(OUT/(name+'.evidence.json'))]
        checked=subprocess.run(command,cwd=REPO,capture_output=True,text=True,encoding='utf8',check=False)
        check={'command':command,'exit_code':checked.returncode,'stdout':checked.stdout,'stderr':checked.stderr}
        save(OUT/(name+'.checker.json'),check);checks[name]=check
        assert (checked.returncode==0)==(result['real_smoke']=='passed')
    assert Counter(measured)==Counter((r['run_id'],r['request_sha256']) for r in smoke)
    assert len(smoke)==freeze['planned_smoke_attempts']
    trace=[]
    for line in (RAW/'model-trace.jsonl').read_text(encoding='utf8').splitlines():
        trace.append(json.loads(line)['data'])
    trace_checks={}
    for name in NAMES:
        config=load_config(OUT/(name+'.smoke.json'));model_id=config.models[0].model_id
        inputs=[r for r in trace if r['type']=='llm.prediction.input' and r.get('modelIdentifier')==model_id]
        outputs=[r for r in trace if r['type']=='llm.prediction.output' and r.get('modelIdentifier')==model_id]
        ready=preflight(config);rows=[json.loads(l) for l in (RAW/(name+'.development-parent.candidate-counts.jsonl')).read_text().splitlines()]
        formats={(r['case_uid'],r['strategy']):r['formatted_sha256'] for r in rows}
        expected=[formats[(j.case.case_id,j.strategy)] for j in ready.jobs]
        actual=[hashlib.sha256(r['input'].encode()).hexdigest() for r in inputs[-len(expected):]]
        formats_match=actual==expected
        counts_match=len(inputs)==len(outputs)==len(expected)+sum(r['model']==name for r in diag)
        if summaries[name]['real_smoke']=='passed':assert formats_match and counts_match,'Actual emitted prompts differ from measured native path: '+name

    save(OUT/'attempt-budget.verification.json',verify_limits())
    blocked={n:r['issue'] for n,r in summaries.items() if r['real_smoke']!='passed'}
    formatted_matches=sum(r['formatted_sha256_match_count'] for r in trace_checks.values());usage_matches=sum(r['usage_match_count'] for r in trace_checks.values())
    validation={'checked_at':datetime.now(UTC).isoformat(),'source_bundle_sha256':compute_source_bundle_hash(),'protected_historical_files_unchanged':len(protected),'candidate_raw_files_unchanged':len(candidate_index),'diagnostic_attempts':len(diag),'smoke_attempts':len(smoke),'total_attempts':len(intents),'generation_attempt_cap':96,'remaining_total_attempts':96-len(intents),'remaining_smoke_attempts':84-len(smoke),'automatic_retries':0,'confirmation_generation_attempts':0,'all_smoke_attempts_from_real_http_adapter':True,'ledger_matches_all_three_production_journals':True,'no_new_runs_or_repeated_smoke_requests':True,'noop_generation_attempts':0,'trace_checks':trace_checks,'formatted_prompt_match_count':formatted_matches,'actual_usage_match_count':usage_matches,'real_smoke':{n:summaries[n]['real_smoke'] for n in NAMES},'infrastructure_blockers':blocked,'ready_for_scale':'blocked','additional_release_blockers':['final executable release not frozen','candidate handoff unverified','historical parent exact source bytes not recovered','GPT dynamic template date not frozen for a multi-day run']}

    scale={'hypothetical_requests':58368,'cases_per_model_strategy':9728,'hardware':'same machine, observed load configs, sequential model/strategy execution','measured_mean_http_seconds':per_scope,'linear_mean_days':mean_seconds/86400,'linear_median_scenario_days':median_seconds/86400,'linear_each_scope_slowest_observed_scenario_days':slow_seconds/86400,'linear_runner_wall_days':end_to_end_seconds/86400,'retries_hypothetical_max3_with_candidate_backoff_2_4_seconds':retries,'actual_pilot_retries':0,'infrastructure_blockers':blocked,'limitations':['11 infrastructure-only cases, deliberately including input maxima; not a representative scientific timing sample','same output length/latency distribution and fixed reasoning/budget assumptions','truncated responses measure capped attempts; latency with adequate future budgets is unknown','no GPU contention, no repeated model loading, no long-run thermal/failure model','runtime/tokenizer/confirmation files remain separately scoped; no scale authorization','tokenization elapsed time is excluded from inference timing']}

    coverage=json.loads((OUT/'coverage.json').read_text())
    lines=['# LM Studio: development infrastructure-smoke v2','',f'База: master после PR11 (`{freeze["source_bundle_sha256"]}` — отдельный source hash исполнявшегося кода; Git-база `e139d67`).', '',f'Выполнено **{len(diag)} диагностических + {len(smoke)} smoke = {len(intents)}/96 реальных generation-попыток**.     for name in NAMES:
        r=summaries[name];cfg=load_config(OUT/(name+'.smoke.json'));errors=', '.join(f'{k}: {v}' for k,v in r['errors'].items()) or '0'

    lines+=['','HTTP completed означает полученный HTTP 200; finalized — сохранённый outcome, включая ошибки и обрезания. Для этих запусков real SmokeEvidence не создан. Бюджеты после ошибок не менялись; новые runs и повторы не выполнялись.     lines+=['', '## Два независимых покрытия', '', '| Модель | Стратегия | Исторические 8, max | Full parent / новый smoke max | Confirmation max | Context / output cap |','|---|---|---:|---:|---:|---:|']

        for strategy in ['direct','z3']:
            c=coverage['models'][name][strategy]

    lines+=['', 'Все 58 368 будущих запросов независимо сверены offline с сохранёнными UID/request digest и длинами при новых предполагаемых development-условиях.
